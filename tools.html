    Read-only EVTX export on Windows
    Export an existing EVTX as structured event XML.
    $sourcePath = 'C:\Evidence\Security.evtx'
$destinationPath = 'C:\Evidence\Security.events.xml'
$events = Get-WinEvent -Path $sourcePath -Oldest -ErrorAction Stop
$eventXml = $events | ForEach-Object { $_.ToXml() }
$xml = '<Events>' + ($eventXml -join [Environment]::NewLine) + '</Events>'
[System.IO.File]::WriteAllText($destinationPath, $xml, [System.Text.UTF8Encoding]::new($false))
